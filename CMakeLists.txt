cmake_minimum_required(VERSION 3.20)
project(ZeroPad VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(ZEROPAD_GUI_SMOKE_TESTS "Build interactive Windows dialog smoke tests using mocked input" OFF)
option(ZEROPAD_PERFORMANCE_PROBE "Build standalone Windows engine performance probe" OFF)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/include/cpptoml.h" CPPTOML_HEADER_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/cpptoml.cpp" CPPTOML_SOURCE_HASH)
if(NOT CPPTOML_HEADER_HASH STREQUAL "e79ec130e4e074c30e7d2103d96390db1566549c6321e00798b8342ea57e6aba"
   OR NOT CPPTOML_SOURCE_HASH STREQUAL "1f474d5942a84b23af547e7e60b8b2bc6ebd54113eed04c710339ffea70c455a")
    message(FATAL_ERROR "Vendored cpptoml differs from the pinned revision; see third_party/cpptoml/PROVENANCE.md")
endif()
add_library(cpptoml STATIC src/cpptoml.cpp)
target_include_directories(cpptoml PUBLIC include)
add_library(zeropad_mapper src/mapper.cpp src/keyboard_state.cpp)
target_include_directories(zeropad_mapper PUBLIC include)
if(MSVC)
    target_compile_options(zeropad_mapper PRIVATE /W4 /permissive-)
else()
    target_compile_options(zeropad_mapper PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(zeropad_config src/config.cpp src/config_io.cpp src/runtime.cpp)
target_link_libraries(zeropad_config PUBLIC zeropad_mapper PRIVATE cpptoml)
if(MSVC)
    target_compile_options(zeropad_config PRIVATE /W4 /permissive-)
else()
    target_compile_options(zeropad_config PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    target_link_libraries(zeropad_config PRIVATE Shell32 Ole32)
endif()
find_package(Threads REQUIRED)
add_library(zeropad_settings src/settings.cpp src/engine_service.cpp src/lifecycle.cpp)
target_link_libraries(zeropad_settings PUBLIC zeropad_config Threads::Threads)
if(MSVC)
    target_compile_options(zeropad_settings PRIVATE /W4 /permissive-)
else()
    target_compile_options(zeropad_settings PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    enable_language(RC)
    add_library(zeropad_windows src/settings_window.cpp src/tray.cpp src/tray_icon.cpp src/single_instance.cpp src/controller.cpp src/keyboard.cpp)
    target_compile_definitions(zeropad_windows PUBLIC UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
    target_include_directories(zeropad_windows PRIVATE resources)
    target_link_libraries(zeropad_windows PUBLIC zeropad_settings Xinput9_1_0 User32 Comctl32 Gdi32 Wtsapi32)
    add_executable(ZeroPad WIN32 src/main.cpp resources/zeropad.rc)
    target_include_directories(ZeroPad PRIVATE resources)
    target_link_libraries(ZeroPad PRIVATE zeropad_windows)
    if(MSVC)
        target_compile_options(ZeroPad PRIVATE /W4 /permissive- /arch:AVX2)
        target_compile_options(zeropad_windows PRIVATE /W4 /permissive- /arch:AVX2)
        target_link_options(ZeroPad PRIVATE /MANIFEST:NO /LTCG)
    endif()
    if(ZEROPAD_PERFORMANCE_PROBE)
        add_executable(performance_probe tools/performance_probe.cpp)
        target_link_libraries(performance_probe PRIVATE zeropad_windows Psapi)
    endif()
endif()
if(BUILD_TESTING)
    add_executable(mapper_tests tests/mapper_tests.cpp)
    target_link_libraries(mapper_tests PRIVATE zeropad_mapper)
    add_test(NAME mapper_tests COMMAND mapper_tests)
    add_executable(config_tests tests/config_tests.cpp)
    target_link_libraries(config_tests PRIVATE zeropad_config)
    add_test(NAME config_tests COMMAND config_tests)
    add_executable(settings_tests tests/settings_tests.cpp)
    target_link_libraries(settings_tests PRIVATE zeropad_settings)
    add_test(NAME settings_tests COMMAND settings_tests)
    add_executable(lifecycle_tests tests/lifecycle_tests.cpp)
    target_link_libraries(lifecycle_tests PRIVATE zeropad_settings)
    add_test(NAME lifecycle_tests COMMAND lifecycle_tests)
    if(WIN32)
        add_executable(native_lifecycle_tests tests/native_lifecycle_tests.cpp)
        target_link_libraries(native_lifecycle_tests PRIVATE zeropad_windows)
        add_test(NAME native_lifecycle_tests COMMAND native_lifecycle_tests)
    endif()
    if(WIN32 AND ZEROPAD_GUI_SMOKE_TESTS)
        add_executable(gui_smoke_tests tests/gui_smoke_tests.cpp resources/zeropad.rc)
        target_include_directories(gui_smoke_tests PRIVATE resources)
        target_compile_definitions(gui_smoke_tests PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
        target_link_libraries(gui_smoke_tests PRIVATE zeropad_windows)
        if(MSVC)
            target_compile_options(gui_smoke_tests PRIVATE /W4 /permissive-)
            target_link_options(gui_smoke_tests PRIVATE /MANIFEST:NO)
        endif()
        add_test(NAME gui_smoke_tests COMMAND gui_smoke_tests)
        set_tests_properties(gui_smoke_tests PROPERTIES TIMEOUT 60 LABELS "interactive")
        add_executable(tray_smoke_tests tests/tray_smoke_tests.cpp resources/zeropad.rc)
        target_include_directories(tray_smoke_tests PRIVATE resources)
        target_link_libraries(tray_smoke_tests PRIVATE zeropad_windows Psapi)
        if(MSVC)
            target_compile_options(tray_smoke_tests PRIVATE /W4 /permissive-)
            target_link_options(tray_smoke_tests PRIVATE /MANIFEST:NO)
        endif()
        add_test(NAME tray_smoke_tests COMMAND tray_smoke_tests)
        set_tests_properties(tray_smoke_tests PROPERTIES TIMEOUT 60 LABELS "interactive")
    endif()
endif()
